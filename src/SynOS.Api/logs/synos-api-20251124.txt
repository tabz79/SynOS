    [ResultId] uniqueidentifier NOT NULL,
    [ParameterCode] nvarchar(50) NOT NULL,
    [ParameterName] nvarchar(200) NOT NULL,
    [Value] decimal(18,4) NOT NULL,
    [CriticalThreshold] nvarchar(50) NOT NULL,
    [PatientId] uniqueidentifier NOT NULL,
    [VisitId] uniqueidentifier NOT NULL,
    [ReferrerId] uniqueidentifier NULL,
    [TriggeredAt] datetimeoffset NOT NULL,
    [NotifiedTo] nvarchar(500) NOT NULL,
    [NotifiedAt] datetimeoffset NULL,
    [AcknowledgedByUserId] uniqueidentifier NULL,
    [AcknowledgedAt] datetimeoffset NULL,
    [AckMethod] nvarchar(50) NULL,
    [AckNotes] nvarchar(max) NULL,
    [EscalatedAt] datetimeoffset NULL,
    [Status] nvarchar(50) NOT NULL,
    [CreatedAt] datetimeoffset NOT NULL,
    CONSTRAINT [PK_CriticalAlerts] PRIMARY KEY ([AlertId]),
    CONSTRAINT [FK_CriticalAlerts_Patients_PatientId] FOREIGN KEY ([PatientId]) REFERENCES [Patients] ([PatientId]) ON DELETE CASCADE,
    CONSTRAINT [FK_CriticalAlerts_Referrers_ReferrerId] FOREIGN KEY ([ReferrerId]) REFERENCES [Referrers] ([ReferrerId]),
    CONSTRAINT [FK_CriticalAlerts_Results_ResultId] FOREIGN KEY ([ResultId]) REFERENCES [Results] ([ResultId]) ON DELETE CASCADE,
    CONSTRAINT [FK_CriticalAlerts_Users_AcknowledgedByUserId] FOREIGN KEY ([AcknowledgedByUserId]) REFERENCES [Users] ([UserId]),
    CONSTRAINT [FK_CriticalAlerts_Visits_VisitId] FOREIGN KEY ([VisitId]) REFERENCES [Visits] ([VisitId]) ON DELETE CASCADE
);
